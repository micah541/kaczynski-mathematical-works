\section{A cube with a non-Borel set of curvilinear convergence}\label{sec:cube}
% ===========================================================================
Source: \cite{PAMS}. For an open connected $\Omega\subseteq\R^n$ and $f:\Omega\to\R^m$, the set of curvilinear convergence $\CC(f)$ is the set of
$p\in\partial\Omega$ for which there is an arc $\gamma$ at $p$ with $f$ converging to a finite limit along $\gamma$.

\begin{definition}[asymptotic values]\label{def:asymptotic}
\lean{TJK.Cube.IsAsymptoticValue, TJK.Cube.IsPointAsymptoticValue}
\uses{def:ccset, cor:path-arc}
$a\in\R^m$ is an \emph{asymptotic value} of $f:\Omega\to\R^m$ if there is a continuous $v:[0,1)\to\Omega$ with $\dist(v(t),\R^n\setminus\Omega)\to0$
and $f(v(t))\to a$ as $t\to1$; it is a \emph{point asymptotic value at $p$} if moreover $v(t)\to p\in\partial\Omega$. By
Corollary~\ref{cor:path-arc}, $p\in\CC(f)$ iff $f$ has a (finite) point asymptotic value at $p$.
\end{definition}

\begin{theorem}[analytic non-Borel sets]\label{fact:analytic-nonborel}
\lean{TJK.Cube.exists_analytic_not_borel, TJK.Cube.exists_analytic_not_borel'}
\source{Souslin--Lusin; Sierpi\'nski, General topology p.~254. Mathlib has MeasureTheory.AnalyticSet}
There is an analytic subset of $\R$ that is not a Borel set.
\end{theorem}
\begin{proof}Formalized in \texttt{TJK/Cube/AnalyticNonBorel.lean}: a universal open set in $(\N\to\{0,1\})\times Y$ built from a countable basis gives, by projecting closed graphs, an analytic set universal for analytic subsets of Cantor space; its diagonal is analytic with non-analytic complement, hence not Borel; transfer to $\R$ by a continuous injection of Cantor space.\end{proof}

\begin{lemma}[{\cite[Lemma]{PAMS}}]\label{lem:cube-lemma}
\lean{TJK.Cube.exists_s}
\uses{lem:cube-lemma-elementary, def:asymptotic}
There is a continuous $s:\Hp\to\C$ with $\abs s<1$ such that the set $E$ of real asymptotic values of $s$ lying in $(-1,1)$ equals the set
of real point asymptotic values of $s$ lying in $(-1,1)$, and $E$ is not a Borel set.
\end{lemma}
\begin{proof}
This is Lemma~\ref{lem:cube-lemma-elementary} below.
\end{proof}

\begin{remark}[Kaczynski's proof]
Kaczynski derives the lemma from two classical theorems, which are not formalized here:
(K) (Kierst 1936) for every analytic $A\subseteq\R$ there is a holomorphic $h$ in $\D$ omitting $i$ and $-i$ whose set of finite
asymptotic values is $A\cup\{-i,i\}$; (N) (Lehto--Virtanen \cite[p.~53]{LV}; Bagemihl--Seidel \cite[Theorem 1]{BS}) if $h$ is
holomorphic in $\D$ and omits two finite values, every finite asymptotic value of $h$ is a point asymptotic value. The argument:
Let $A$ be analytic and not Borel, $h$ as in Kierst's theorem (K), and $\rho(w)=w/(1+\abs w)$, a homeomorphism of $\C$ onto $\D$ mapping
$\R$ onto $(-1,1)$, with inverse $u\mapsto u/(1-\abs u)$. Put $s(x,y)=\rho\bigl(h(\max\{1-y,0\}e^{ix})\bigr)$, continuous on $\Hp$
with $\abs s<1$. Let $b\in(-1,1)$. If $b$ is an asymptotic value of $s$ along $v=\langle x(t),y(t)\rangle$, then $y(t)=\dist(v(t),\C\setminus\Hp)\to0$, and
$\rho^{-1}(b)$ is an asymptotic value of $h$ along $w=(1-y)e^{ix}$ (for $y<1$, $\dist(w,\C\setminus\D)\le\abs{w-e^{ix}}=y$); as $\rho^{-1}(b)$ is real,
it lies in $A$. If $a\in A$, by (N) $a$ is a point asymptotic value of $h$ at some $e^{ip}$, $p=\arg e^{ip}$, along a path
$w(t)\to e^{ip}$. For $t$ near $1$, $w(t)e^{-ip}$ lies near $1$, off the negative axis, so $\vartheta(t)=p+\arg(w(t)e^{-ip})$ is continuous,
$\vartheta(t)\to p$ and $\abs{w(t)}e^{i\vartheta(t)}=w(t)$. Hence $\rho(a)$ is a point asymptotic value of $s$ at $p$ along
$\langle\vartheta(t),1-\abs{w(t)}\rangle$. Since point asymptotic values are asymptotic values, both sets equal $\rho(A)$, which is not Borel
($A=\rho^{-1}(\rho(A))$ and $\rho$ is continuous).
\end{remark}

\begin{remark}
\cite{PAMS} notes that the lemma can also be proved without meromorphic functions, from \cite[Theorem 113]{SierpGT}, a theorem of
Bagemihl--McMillan, and the methods of \cite{TAMSb}, but omits the ``messy construction''.
\end{remark}

\subsection*{An elementary proof of the lemma}
We give a proof of Lemma~\ref{lem:cube-lemma} that uses neither Kierst's theorem nor normal functions: only
Theorem~\ref{fact:analytic-nonborel} and plane geometry. The function $s$ is built on a binary tree of segments in $\Hp$
(a ``dyadic skeleton''); its imaginary part forces every path along which $s$ tends to a real value to follow one branch of the
tree, and its real part is read off a Souslin representation of the analytic set along the branches.

\emph{Words and the skeleton.} Let $W$ be the set of finite words $u=u_0\dots u_{n-1}$ over $\{0,1\}$; $\abs u=n$, $u\sqsubseteq v$
means $u$ is a prefix of $v$, $u\sqsubset v$ a proper prefix, $u|k$ is the prefix of length $k$, and $u^-$ is $u$ without its last
letter ($u\ne\emptyset$). The same notation is used for infinite words $\beta\in\{0,1\}^\N$. Put $x_u=\sum_{i<\abs u}u_i2^{-i-1}$,
$I_u=[x_u,x_u+2^{-\abs u}]$, $\mu_u=x_u+2^{-\abs u-1}$ (the midpoint of $I_u$), and $c_u=\mu_u+i\,2^{-\abs u}\in\Hp$. For
$u\ne\emptyset$ let $e_u$ be the segment $[c_{u^-},c_u]$, and for $z\in\Hp$ let $d_u(z)=\dist(z,e_u)/\Im z$.

\begin{enumerate}
\item[(G1)] Points of $e_u$ have heights in $[2^{-\abs u},2^{1-\abs u}]$.
\item[(G2)] If $\abs v>k$ and $p\in e_v$, then $\Re p\in I_{v|k}$ and $\dist(\Re p,2^{-k}\Z)\ge\Im p/4$. Indeed, let $j=\abs v$.
The endpoints of $e_v$ have real parts $\mu_{v^-}$, an odd multiple of $2^{-j}$, at distance $2^{-j}$ from $2^{1-j}\Z\supseteq2^{-k}\Z$,
and $\mu_v$, an odd multiple of $2^{-j-1}$, at distance $2^{-j-1}$ from $2^{-j}\Z\supseteq2^{-k}\Z$; both lie in $I_v\subseteq I_{v|k}$.
On $I_{v|k}$ the function $x\mapsto\dist(x,2^{-k}\Z)$ is the minimum of the distances to the two endpoints, hence concave, so along
$e_v$ (on which $\Re$ is affine) it is at least $2^{-j-1}\ge\Im p/4$ by (G1).
\item[(G3)] If $u1\sqsubseteq v$ and $p\in e_v$ then $\Re p\ge\mu_u$; if $u0\sqsubseteq v$ then $\Re p\le\mu_u$. (For $v=u1$ the
endpoints of $e_v$ have real parts $\mu_u<\mu_{u1}$; for $v\sqsupset u1$ both endpoints lie in $I_{u1}=[\mu_u,\mu_u+2^{-\abs u-1}]$.
Symmetrically for $u0$.)
\item[(G4)] \emph{Separation.} If $p,p'\in\Hp$, $q\in\R$, $\Re p\ge q$ and $\Re p'\le q-\Im p'/4$, then
$\abs{p-p'}\ge\max\{\Im p,\Im p'\}/8$. If $\Im p'\ge\frac78\Im p$ then $\abs{p-p'}\ge\Re p-\Re p'\ge\Im p'/4\ge\max\{\Im p,\Im p'\}\cdot\frac7{32}$;
otherwise $\abs{p-p'}\ge\Im p-\Im p'>\Im p/8$ and $\Im p=\max\{\Im p,\Im p'\}$.
\item[(G5)] Call $e_v,e_w$ \emph{separated} if $\abs{p-p'}\ge\max\{\Im p,\Im p'\}/8$ for all $p\in e_v$, $p'\in e_w$. This holds
(a) if $\abs v,\abs w>k$ and $v|k\ne w|k$: if, say, $I_{v|k}$ lies to the left of $I_{w|k}$, let $q$ be the right endpoint of $I_{v|k}$; by
(G2), $\Re p\le q-\Im p/4$ and $\Re p'\ge q$; apply (G4). (b) if $u1\sqsubseteq v$ and $u0\sqsubset w$: take $q=\mu_u$, the right endpoint
of $I_{u0}$, and use (G3) for $v$ and (G2) with $k=\abs u+1$ for $w$.
\item[(G6)] \emph{Nearness.} If $e_v,e_w$ are separated and $d_v(z)<\frac1{32}$, then $d_w(z)>\frac1{12}$. Take $p\in e_v$ with
$\abs{z-p}<\Im z/32$; then $\Im p>\frac{31}{32}\Im z$, so every $p'\in e_w$ has $\abs{z-p'}\ge\abs{p-p'}-\abs{z-p}>(\frac{31}{256}-\frac8{256})\Im z>\Im z/12$.
\item[(G7)] If $\Im z<2^{-k-1}$ and $\abs w\le k$, then $d_w(z)\ge1$, by (G1).
\end{enumerate}

\emph{Coding.} Write $\N^\N$ for the Baire space and $N_\kappa=\{\sigma\in\N^\N:\kappa\sqsubseteq\sigma\}$ for $\kappa\in\N^{<\N}$.
Every $u\in W$ can be written uniquely as $u=0^{k_0}1\,0^{k_1}1\cdots0^{k_{j-1}}1\,0^r$; put $\kappa(u)=(k_0,\dots,k_{j-1})$ and
$\ell(u)=\abs u-r$ (the length of the longest prefix of $u$ ending in $1$). Then $u\sqsubseteq v$ implies $\kappa(u)\sqsubseteq\kappa(v)$, and
$\ell(u)\ge\abs{\kappa(u)}$, the number of $1$'s in $u$. For $\sigma\in\N^\N$ let
$\beta_\sigma=0^{\sigma_0}1\,0^{\sigma_1}1\cdots$; then $\kappa(\beta_\sigma|m)\sqsubseteq\sigma$ and $\abs{\kappa(\beta_\sigma|m)}\to\infty$. If
$\beta\in\{0,1\}^\N$ has infinitely many $1$'s, then $\beta=\beta_\sigma$ for some $\sigma$ (the lengths of the $0$-blocks); otherwise
$\beta=t0^\infty$ with $t$ empty or ending in $1$.

\begin{lemma}[elementary form of Lemma~\ref{lem:cube-lemma}]\label{lem:cube-lemma-elementary}
\lean{TJK.Cube.Elementary.exists_s'}
\uses{fact:analytic-nonborel, def:asymptotic}
The conclusion of Lemma~\ref{lem:cube-lemma} holds, with a proof using only Theorem~\ref{fact:analytic-nonborel}.
\end{lemma}
\begin{proof}
Let $A_0\subseteq\R$ be analytic and not Borel, and $A=\varphi(A_0)$ with $\varphi(a)=a/(2(1+\abs a))$, a continuous injection of $\R$
into $(-\frac12,\frac12)$. $A$ is analytic, and not Borel because $A_0=\varphi^{-1}(A)$. As $A\neq\emptyset$, $A=g(\N^\N)$ for a continuous
$g:\N^\N\to\R$. For $u\in W$ let $r_u=g(\kappa(u)0^\infty)\in g(N_{\kappa(u)})\subseteq(-\frac12,\frac12)$ and $\psi_u=1/(\ell(u)+2)$. For $z\in\Hp$ put
\[R(z)=\min\Bigl\{\tfrac12,\ \inf_{u\ne\emptyset}\bigl(r_u+12\,d_u(z)\bigr)\Bigr\},\qquad
\Psi(z)=\min\Bigl\{1,\ \inf_{u\ne\emptyset}\bigl(\psi_u+16\,d_u(z)\bigr)\Bigr\},\qquad s=R+\tfrac i2\Psi.\]
\emph{Continuity and bounds.} On $K=\{z:\Im z\ge\eta,\ \abs z\le M\}$ each $z\mapsto c_u+C\dist(z,e_u)/\Im z$ (with $\abs{c_u}\le1$,
$e_u\subseteq[0,1]\times[0,1]$) is Lipschitz with a constant $C(1/\eta+(M+2)/\eta^2)$ independent of $u$, and the infimum of a family of
uniformly Lipschitz functions bounded below is Lipschitz with the same constant; so $R,\Psi$, hence $s$, are continuous on $\Hp$.
As $r_u>-\frac12$, $R\in[-\frac12,\frac12]$; $\Psi\in[0,1]$; so $\abs s\le\sqrt{1/4+1/4}<1$.

\emph{Claim L.} Let $\abs u=k$, $\Im z<2^{-k-1}$ and $d_v(z)<\frac1{32}$ for some $v\sqsupset u$. Then every $w$ with $d_w(z)\le\frac1{12}$
satisfies $w\sqsupset u$, and $R(z)\ge\min\{\frac12,\inf_{w\sqsupset u}r_w\}$. Indeed if $w\not\sqsupset u$ then either $\abs w\le k$ and
$d_w(z)\ge1$ by (G7), or $\abs w>k$, $w|k\ne u$, and $d_w(z)>\frac1{12}$ by (G5a), (G6). Then $r_w+12d_w(z)>-\frac12+1=\frac12$.

\emph{Point asymptotic values.} Let $a=g(\sigma)\in A$, $\beta=\beta_\sigma$, $p=\sum_i\beta_i2^{-i-1}=\lim\mu_{\beta|m}$. With
$\rho_m(y)=\min\{1,\max\{0,2-2^my\}\}$ let $X(y)=\mu_\emptyset+\sum_{m\ge1}(\mu_{\beta|m}-\mu_{\beta|m-1})\rho_m(y)$, a uniformly convergent
series of continuous functions ($\abs{\mu_{\beta|m}-\mu_{\beta|m-1}}=2^{-m-1}$). For $y\in[2^{-m},2^{1-m}]$, $\rho_j(y)=1$ for $j<m$ and
$\rho_j(y)=0$ for $j>m$, so with $\lambda=2-2^my\in[0,1]$, $z(y)=X(y)+iy=(1-\lambda)c_{\beta|m-1}+\lambda c_{\beta|m}\in e_{\beta|m}$. Hence
$X(y)\to p$ as $y\to0$, and $d_{\beta|m}(z(y))=0$, so $\Psi(z(y))\le\psi_{\beta|m}\to0$ (as $\ell(\beta|m)\ge\abs{\kappa(\beta|m)}\to\infty$)
and $R(z(y))\le r_{\beta|m}\to a$ (as $\kappa(\beta|m)0^\infty\to\sigma$). Given $\eps>0$ choose $j$ with $\abs{g-a}<\eps$ on $N_{\sigma|j}$
and $k$ with $\abs{\kappa(\beta|k)}\ge j$; for $w\sqsupset\beta|k$, $r_w\in g(N_{\sigma|j})$, so Claim L gives $R(z(y))\ge a-\eps$ for
$y<2^{-k-1}$. So $s\to a$ along $t\mapsto z(1-t)$, and $a$ is a point asymptotic value of $s$ at $p$.

\emph{Asymptotic values.} Let $s\to b\in\R$ along a path $v:[0,1)\to\Hp$ with $\Im v(t)\to0$. Then $\Psi(v(t))\to0$ and $R(v(t))\to b$. For
$\abs u=k$ let $\Omega_k(u)=\{z:\Im z<2^{-k-1},\ \exists v\sqsupset u,\ d_v(z)<\frac1{32}\}$, an open set. By Claim L the $\Omega_k(u)$,
$\abs u=k$, are pairwise disjoint, and every $z$ with $\Im z<2^{-k-1}$ and $\Psi(z)<\frac12$ lies in one of them (some $w$ has
$d_w(z)<\frac1{32}$; then $\abs w>k$ by (G7); take $u=w|k$). So for each $k$ a tail $v([t_k,1))$, being connected, lies in a single
$\Omega_k(u_k)$; a common point of two tails and Claim L give $u_k\sqsubseteq u_{k+1}$, so $u_k=\beta|k$ for some $\beta\in\{0,1\}^\N$.
By (G2), points of $\Omega_k(u)$ have real part within $2^{-k-1}+\Im z/32$ of $\mu_u$, so $v(t)\to p=\lim\mu_{\beta|k}\in\R$: $b$ is a point
asymptotic value of $s$ at $p$. It remains to show $b\in A$.

\emph{Case 1: $\beta=\beta_\sigma$.} Let $a=g(\sigma)$, and $\eps$, $k$ as above. For $t\ge t_k$ with $\Psi(v(t))<\delta<\frac12$ there is $w$
with $\psi_w+16d_w(v(t))<\delta$; then $d_w<\frac1{32}$, so $w\sqsupset\beta|k$ by Claim L and $R(v(t))\le r_w+12d_w<a+\eps+\delta$, while
$R(v(t))\ge a-\eps$. Hence $b=a\in A$.

\emph{Case 2: $\beta=t0^\infty$}, $t$ empty or ending in $1$, $n=\abs t$, so $\psi_{t0^i}=c_0:=1/(n+2)$ for all $i$. Fix $k\ge n$. For $t$ large
($t\ge t_k$ and $\Psi(v(t))<c_0/2$) there is $w$ with $\psi_w<c_0/2$ and $d_w(v(t))<\frac1{32}$; so $w\sqsupset\beta|k=t0^{k-n}$ (Claim L)
and $w$ is not of the form $t0^i$; hence $t0^i1\sqsubseteq w$ for some $i\ge k-n$. Thus a tail of $v$ lies in $\bigcup_iS_i$, where
$S_i=\{z\in\Hp:\exists w\sqsupseteq t0^i1,\ d_w(z)<\frac1{32}\}$ is open. For $i<i'$, $S_i\cap S_{i'}=\emptyset$ by (G5b) (with $u=t0^i$,
$t0^{i+1}\sqsubset t0^{i'}1$) and (G6). So the tail lies in a single $S_i$. But the tail after $t_{n+i+1}$ also lies in
$\Omega_{n+i+1}(t0^{i+1})$, and a point of both is within $\Im z/32$ of some $e_w$, $w\sqsupseteq t0^i1$, and of some $e_{w'}$,
$w'\sqsupset t0^{i+1}$, contradicting (G5b) and (G6). So Case 2 does not occur.

Thus the real asymptotic values of $s$ are exactly the points of $A\subseteq(-\frac12,\frac12)$, all of them point asymptotic values; and
$A$ is not Borel.
\end{proof}

\begin{remark}
In Case 2 the imaginary part does the work: along the branch $t0^\infty$ (the only kind of branch not coding a point of $\N^\N$)
$\Psi$ stays $\ge c_0$ near the branch, and leaving it means entering one of the disjoint ``side'' regions $S_i$. Paths tending to
$\infty$ or oscillating along $\R$ are excluded because $\Psi\to0$ forces the path into the regions $\Omega_k(u)$, whose real parts
are confined to $I_u$ up to $\Im z/32$.
\end{remark}

\begin{theorem}[{\cite[Theorem]{PAMS}}]\label{thm:cube}
\lean{TJK.Cube.exists_not_measurableSet_curvConvSet}
\uses{lem:cube-lemma, thm:construct-C, lem:subarc, cor:path-arc, def:asymptotic}
There is a bounded continuous $f:K^\circ\to\C$ on the open cube $K^\circ=(-1,1)\times(0,1)\times(-1,1)$ whose set of curvilinear convergence
is not a Borel set. (So McMillan's theorem does not extend to three dimensions.)
\end{theorem}
\begin{proof}
Let $s,E$ be as in Lemma~\ref{lem:cube-lemma}, $g(x,y)=s(x/y,y)$ for $y>0$, so $\abs g<1$.

\emph{Construction.} For $t\in\R$ let $K_t=[-1,1]\times[t,1]$ and let $d_0(t)$ be the supremum of the $\delta\in[0,1]$ such that $z,z'\in K_t$,
$\abs{z-z'}<\delta$ imply $\abs{g(z)-g(z')}\le t$. Raising $t$ shrinks $K_t$ and loosens the tolerance, so $d_0$ is weakly increasing; for $t>0$,
$d_0(t)>0$ because $g$ is uniformly continuous on the compact set $K_t$. Let $d(t)=\min\{\frac12d_0(t/2),\frac14t\}$, positive and weakly increasing
on $(0,1]$, and $k$ as in Theorem~\ref{thm:construct-C} (so $\abs k\le\sqrt2\le2$). Put $f(x,y,z)=(g(x,y)-z)\,k(x,y)$, continuous on $K^\circ$ with
$\abs f\le4$. For $c,e',e'',k',k''\in\C$ with $\abs{k'},\abs{k''}\le2$,
\begin{equation}\label{eq:cube1}
\abs{(c+e')k'-(c+e'')k''}\ge\abs c\abs{k'-k''}-2\abs{e'}-2\abs{e''},
\end{equation}
because $(c+e')k'-(c+e'')k''=c(k'-k'')+e'k'-e''k''$.

Let $F=\CC(f)$. We show that for $b\in(-1,1)$, $\langle0,0,b\rangle\in F$ iff $b\in E$; then $F$ is not Borel, since otherwise
$\{b\in(-1,1):\langle0,0,b\rangle\in F\}=E$ would be.

If $b\in E$, $s\to b$ along a path $\langle u(t),v(t)\rangle\to\langle p,0\rangle$ in $\Hp$. Along $c(t)=\langle u(t)v(t),v(t),b\rangle\to\langle0,0,b\rangle$,
which lies in $K^\circ$ for $t$ near $1$, $g=s(u,v)\to b$, so $f\to0$. By Corollary~\ref{cor:path-arc} (applied to $c$ restricted to some $[t_0,1)$
and closed up by $\langle0,0,b\rangle$, which lies in $\partial K^\circ$), $\langle0,0,b\rangle\in F$.

Conversely let $f\to q$ along an arc $\gamma'=q_0([0,1])$ at $q_0(0)=\langle0,0,b\rangle$ in $K^\circ$. We claim $g(x,y)-z\to0$ along $\gamma'$.
Then $g\to b$ along $\gamma'$, so $s\to b$ along $v(t)=\langle x/y,y\rangle(q_0(1-t))$, $t\in[0,1)$, a path in $\Hp$ whose second coordinate tends
to $0$: $b$ is an asymptotic value of $s$, so $b\in E$.

If the claim fails, there is $\eps>0$ such that $\abs{g(x,y)-z}\ge\eps$ at points of $\gamma'\setminus\{\langle0,0,b\rangle\}$ arbitrarily close to $\langle0,0,b\rangle$.
Choose $r_1>0$ with $\abs{f(P)-q}<\eps/2$ for $P\in\gamma'\setminus\{\langle0,0,b\rangle\}$ within $r_1$ of $\langle0,0,b\rangle$, and put $r=\min\{r_1,\eps/16\}$. As $\gamma'$
is compact and its only point with $y=0$ is $\langle0,0,b\rangle$, there is $\delta>0$ such that points of $\gamma'$ with $y\le\delta$ lie within $r$ of
$\langle0,0,b\rangle$. Take $P_n=q_0(t_n)=\langle x_n,y_n,z_n\rangle$ with $\abs{g(x_n,y_n)-z_n}\ge\eps$ and $0<y_n<\min\{3\delta/4,3\eps/32,3/4\}$; put
$a=4y_n/3\le1$, so $a\le\delta$ and $a<\eps/8$. The projection $\Phi$ of $q_0$ to the $\langle x,y\rangle$-plane has $\Phi(0)=\langle0,0\rangle$ at distance
$\ge y_n>d(a)$ from $\Phi(t_n)$, so by Lemma~\ref{lem:subarc}(b) there is an interval $J\ni t_n$ with $\gamma^*=\Phi(J)$ in the closed disc of radius
$d(a)\le a/4$ about $\langle x_n,y_n\rangle$ and containing a point at distance $d(a)$ from it. Then $\gamma^*\subseteq[-1,1]\times[a/2,a]$
is connected, and Theorem~\ref{thm:construct-C} gives $\tau',\tau''\in J$ with $\abs{k'-k''}\ge2$ at $\Phi(\tau'),\Phi(\tau'')$. The points
$P_n'=q_0(\tau')$, $P_n''=q_0(\tau'')$ have $a/2\le y\le a\le\delta$, so they differ from $\langle0,0,b\rangle$ and lie within $r$ of it; hence
$\abs{f(P_n')-f(P_n'')}<\eps$, $\abs{z_n'-z_n},\abs{z_n''-z_n}<2r\le\eps/8$, and, as $\Phi(\tau'),\langle x_n,y_n\rangle\in K_{a/2}$ are closer than
$d(a)<d_0(a/2)$, $\abs{g'-g(x_n,y_n)},\abs{g''-g(x_n,y_n)}\le a/2<\eps/16$. Applying \eqref{eq:cube1} with $c=g(x_n,y_n)-z_n$, $c+e'=g'-z_n'$,
$c+e''=g''-z_n''$ gives $\eps>\abs{f(P_n')-f(P_n'')}\ge2\eps-4(\eps/16+\eps/8)=\frac54\eps$, a contradiction.
\end{proof}

\begin{remark}
\cite{PAMS} leaves open whether a continuous \emph{real}-valued function in a cube can have a non-Borel set of curvilinear convergence.
\end{remark}
